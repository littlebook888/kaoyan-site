## 3. Supabase 表结构（追加到主站 schema）

```sql
-- 题库进度：选择/提交/笔记，按用户隔离（演示用单用户模型，同主站）
create table if not exists tiku_progress (
  id uuid primary key default gen_random_uuid(),
  user_id text not null default 'kaoyan_user_default',
  group_storage_id text not null,
  subject text not null default '',
  selections jsonb not null default '{}'::jsonb,
  submitted boolean not null default false,
  notes text,
  updated_at timestamptz not null default now(),
  unique(user_id, group_storage_id)
);
create index if not exists idx_tiku_progress_user on tiku_progress(user_id);
alter table tiku_progress enable row level security;
drop policy if exists "tiku_progress_anon" on tiku_progress;
create policy "tiku_progress_anon" on tiku_progress for all
  using (user_id = 'kaoyan_user_default') with check (user_id = 'kaoyan_user_default');

create table if not exists tiku_favorites (
  id uuid primary key default gen_random_uuid(),
  user_id text not null default 'kaoyan_user_default',
  group_storage_id text not null,
  subject text,
  created_at timestamptz not null default now(),
  unique(user_id, group_storage_id)
);
alter table tiku_favorites enable row level security;
drop policy if exists "tiku_favorites_anon" on tiku_favorites;
create policy "tiku_favorites_anon" on tiku_favorites for all
  using (user_id = 'kaoyan_user_default') with check (user_id = 'kaoyan_user_default');
```

> 注：上面的 `user_id` 用 `text` 而非 `auth.users` FK，是为了与主站现有"单用户 anon 演示"模型完全一致（主站 `sql/schema.sql` 即此模式）。若日后要真多账号，再迁到 `auth.users`。
